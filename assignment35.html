/* WHOLE PAGE IS FLEXBOX - No Grid */ /* Navbar */ .navbar { display:flex; justify-content:space-between; align-items:center; flex-wrap:wrap; } /* Hero - Two columns */ .hero { display:flex; flex-direction:row; justify-content:space-between; align-items:center; gap:30px; min-height:80vh; } .hero-content { flex:1 1 50%; display:flex; flex-direction:column; gap:14px; } .hero-visual { flex:1 1 45%; display:flex; justify-content:center; align-items:center; } /* Features - Wrap */ .feature-list { display:flex; flex-wrap:wrap; gap:18px; justify-content:center; } .feature-card { flex:1 1 250px; max-width:320px; } /* Products - Wrap like Q33 */ .product-grid { display:flex; flex-wrap:wrap; gap:18px; justify-content:flex-start; align-items:stretch; } .p-card { flex:1 1 250px; max-width:320px; min-width:240px; } /* Footer - Wrap */ .footer-inner { display:flex; flex-wrap:wrap; gap:30px; justify-content:space-between; } .footer-col { flex:1 1 200px; } @media(max-width:900px){ .hero{flex-direction:column;} }
