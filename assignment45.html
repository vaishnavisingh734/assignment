Q45 - Final Fully Responsive Landing Page - Viva Note

/* Q44 - Only Desktop Base (No media queries) */ .hero { max-width:1200px; display:grid; grid-template-columns:1.2fr 0.8fr; padding:60px 40px; } .hero h1 { font-size:42px; } .feature-grid { grid-template-columns:repeat(3, 1fr); } /* Desktop 3 cols */ .product-grid { grid-template-columns:repeat(4, 1fr); } /* Desktop 4 cols */ .footer { grid-template-columns:repeat(4, 1fr); } /* Q45 - Added Breakpoints to make Q44 Fully Responsive */ /* BREAKPOINT 1 - 1024px Tablet Landscape */ @media(max-width:1024px){ .hero { grid-template-columns:1fr; text-align:center; padding:40px 24px; } .hero h1 { font-size:36px; } /* 42→36 */ .feature-grid { grid-template-columns:repeat(2, 1fr); } /* 3→2 cols */ .product-grid { grid-template-columns:repeat(2, 1fr); } /* 4→2 cols */ .footer { grid-template-columns:repeat(2, 1fr); } /* 4→2 cols */ } /* BREAKPOINT 2 - 768px Mobile (Required) */ @media(max-width:768px){ .navbar { flex-direction:column; align-items:flex-start; } /* Navbar stack */ .hero h1 { font-size:28px; } /* 36→28 */ .feature-grid { grid-template-columns:1fr; } /* 2→1 col */ .footer { grid-template-columns:1fr; } /* 2→1 col */ .features, .products { padding:30px 16px; } /* Padding reduce */ } /* BREAKPOINT 3 - 480px Small Mobile (Required second breakpoint) */ @media(max-width:480px){ .hero h1 { font-size:24px; } /* 28→24 */ .product-grid { grid-template-columns:1fr; } /* 2→1 col */ .hero { padding:24px 12px; } } /* TESTING: Desktop 1200px: hero 1.2fr 0.8fr, features 3 cols, products 4 cols, footer 4 cols, h1 42px Tablet 1024px: hero 1fr stacked, features 2 cols, products 2 cols, footer 2 cols, h1 36px Mobile 768px: navbar column, features 1 col, products 2 cols, footer 1 col, h1 28px Small 480px: products 1 col, h1 24px, compact padding No horizontal scroll, images width 100% height auto, live breakpoint bar at bottom */
